Para que este mismo código modele una enfermedad frecuensodepenendiente se tiene que modificar beta

parameters <- c( beta=0.8, sigma=0.2, gamma=0.1, N= 1000)
parameters
##  beta sigma gamma     N 
## 8e-01 2e-01 1e-01 1e+03
seir_model <- function(time, state, parameters){
   with(as.list(c(state,parameters)), {
     dS <- -beta * S * I / N
     dE <- beta * S * I / N - sigma * E
     dI <- sigma * E  - gamma * I
     dR <- gamma * I
     
  list(c(dS,dE,dI,dR))
  })
}
